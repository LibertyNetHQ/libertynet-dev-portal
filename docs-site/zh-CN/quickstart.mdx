```typescript TypeScript
const res = await fetch("https://registry.libertynet.ai/health");
```

```python Python
import json, urllib.request
with urllib.request.urlopen("https://registry.libertynet.ai/health") as r:
```

```json 返回（一个节点）
  "did": "did:svrp:df9d4b9f390bc49b2210eca81ca3c54c4d63b37b739442a21e28f2ed5b8aa02d",

  "status": "active",

```

<Warning>
**`status: "active"` 不等于在线。**

一个上周就拔了电的节点，这个字段永远还是 `"active"`，没有任何机制会把它改回去。唯一会
"过期"的字段是 `last_seen`——新鲜度得你自己算，第 3 步会正确地做这件事。

**这是新手在这个网络上最常犯的第一个错误。**
</Warning>

## 3. 自己验证一个身份

这一步是 LibertyNet 与众不同的地方，值得花两分钟。

LibertyNet 的 DID 是**自证的**：标识符由公钥推导而来。给你一个公钥和一个声称的身份，
你只用一个 SHA-256 就能确认这对关系——不查目录，不问证书机构，也不必相信刚收到的这个响应。

存成 `verify.py` 跑一下。**不用装任何包。**

```typescript TypeScript
import { createHash } from "node:crypto";

const B58 = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";

function b58decode(s) {
  let n = 0n;
  for (const c of s) {
    const i = B58.indexOf(c);
    if (i < 0) return null;
    n = n * 58n + BigInt(i);
  }
  let hex = n.toString(16);
  if (hex.length % 2) hex = "0" + hex;
  // base58 会丢掉前导零字节，每个前导 '1' 补回一个
  let lead = 0;
  for (const c of s) { if (c === "1") lead++; else break; }
  return Buffer.concat([Buffer.alloc(lead), Buffer.from(hex, "hex")]);
}

// /nodes 给的公钥是 hex，/peers 给的是 base58。同一把钥匙，两种写法。
function keyBytes(pk) {
  return /^[0-9a-f]{64}$/.test(pk) ? Buffer.from(pk, "hex") : b58decode(pk);
}

/** 这个 DID 真的能从这把公钥推导出来吗？ */
function verifyIdBinding(did, publicKey) {
  const key = keyBytes(publicKey);
  if (!key || key.length !== 32) return false;

  const digest = createHash("sha256").update(key).digest("hex");
  const body = did.replace(/^did:svrp:(?:[a-z]:)?/, "");

  if (body.length === 64) return body === key.toString("hex"); // 全 hex 形式
  if (body.length === 8) return body === digest.slice(0, 8);   // 短形式
  if (body.length === 10) return body === digest.slice(0, 10); // 撞名回退形式
  return false;
}

const { nodes } = await (await fetch("https://registry.libertynet.ai/nodes")).json();

let verified = 0;
for (const n of nodes) {
  if (verifyIdBinding(n.did, n.public_key)) verified++;
  else console.log("REJECT", n.did);
}
console.log(`${verified}/${nodes.length} 个身份验证通过`);

const fresh = nodes.filter(
  (n) => n.last_seen && Date.now() - Date.parse(n.last_seen) < 10 * 60 * 1000,
);
console.log(`最近 10 分钟内出现过的有 ${fresh.length} 个`);
```

```python verify.py
import hashlib, json, re, urllib.request

B58 = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"

def b58decode(s):
    n = 0
    for c in s:
        i = B58.find(c)
        if i < 0:
            return None
        n = n * 58 + i
    raw = n.to_bytes((n.bit_length() + 7) // 8, "big")
    # base58 会丢掉前导零字节，每个前导 '1' 补回一个
    return b"\0" * (len(s) - len(s.lstrip("1"))) + raw

# 注册表在 /nodes 用 hex 发公钥，在 /peers 用 base58。是同一个密钥。
def key_bytes(pk):
    if re.fullmatch(r"[0-9a-f]{64}", pk):
        return bytes.fromhex(pk)
    return b58decode(pk)

def verify_id_binding(did: str, public_key: str) -> bool:
    """这个 DID 真的是由这个公钥推导出来的吗？"""
    key = key_bytes(public_key)
    if not key or len(key) != 32:
        return False

    digest = hashlib.sha256(key).hexdigest()
    body = re.sub(r"^did:svrp:(?:[a-z]:)?", "", did)

    if len(body) == 64:
        return body == key.hex()        # 全 hex 形式
    if len(body) == 8:
        return body == digest[:8]       # 短形式
    if len(body) == 10:
        return body == digest[:10]      # 碰撞回退形式
    return False

with urllib.request.urlopen("https://registry.libertynet.ai/nodes", timeout=10) as r:
    nodes = json.load(r)["nodes"]

verified = sum(verify_id_binding(n["did"], n["public_key"]) for n in nodes)
print(f"{verified}/{len(nodes)} 个身份通过验证")
```

```text 真实运行结果，2026-07-31
27/27 个身份通过验证
```

你刚刚在没有向任何人申请许可的情况下，用密码学核对了这个网络上的每一个身份。
**这就是全部的意义所在。**

<Note>
**为什么有两种 DID 编码？** 同一个密钥会以短形式 `did:svrp:n:<8位hex>`（公钥 SHA-256 的截断）
和全形式 `did:svrp:<64位hex>`（公钥本身）两种样子出现。绑定里存短的，运行中的守护进程报全的。
**它们是同一个节点。** 在这里做字符串比较会骗你——详见
[身份](/concepts/identity)。
</Note>

## 4. 调用一个节点，并确认它真的回应了

验证身份只能证明 DID 与公钥对得上。它**不能**证明有人在线，也不能证明回应你的那一方真的持有私钥。这一步两件事都做到。

<Warning>
**网络上大多数节点你根本调用不了。**

很多记录登记的是内网地址或只对运营者有意义的本地标签——对他们是真的，从别处根本连不上。直接向注册表要真正可用的那批：

```bash
curl -s 'https://registry.libertynet.ai/nodes?callable=1'
```

它只返回**公网可达且带注册签名**的节点。今天这个数字很小，SDK 默认就用这一批。
</Warning>

**nonce 由你来定。** 这正是让回应成为「证明」而不是「录音」的关键：对早就准备好的内容的签名，可以直接重放给你。

```python act.py
import json, urllib.request, secrets

def post(url, data):
    req = urllib.request.Request(url, data=json.dumps(data).encode(),
                                 headers={"content-type": "application/json"}, method="POST")
    with urllib.request.urlopen(req, timeout=15) as r:
        return json.load(r)

with urllib.request.urlopen("https://registry.libertynet.ai/nodes?callable=1") as r:
    node = json.load(r)["nodes"][0]

nonce = secrets.token_hex(8)
reply = post(node["endpoint"] + "/echo", {"nonce": nonce})

assert reply["nonce"] == nonce
assert reply["public_key"] == node["public_key"]

print("called   ", node["endpoint"])
print("signature", reply["signature"][:32], "…")
```

现在验证这个签名。Ed25519 需要一个库——这是快速上手里唯一需要装东西的一步：

```bash
pip install cryptography
```

```python verify_reply.py
from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PublicKey

Ed25519PublicKey.from_public_bytes(key_bytes(reply["public_key"])).verify(
    b58decode(reply["signature"]),
    reply["signed_bytes"].encode(),
)
print("ok")
```

跑通了，就说明这个节点确实持有你刚才验证过的那个身份的私钥。

<Warning>
**永远不要把私钥放进配置文件、`.env`、或者仓库里的任何地方。**

这份快速上手全程都不需要私钥——发现和验证都是公开的，demo 节点用的是它自己的密钥，
不是你的。等到你真的持有密钥（作为运营者）时，请在用到的那一刻从操作系统钥匙串或
密钥管理器里读：

```python
import keyring
secret = bytes.fromhex(keyring.get_password("libertynet", "device-key"))
```

放进文件里的密钥，同时也在你的备份里、shell 历史里、崩溃转储里。这个网络**没有重置
流程**——因为重置流程本身就是第二条、而且更弱的入口。
</Warning>

## 5. 或者用 SDK 一行搞定

上面这些就是 SDK 替你做的事，而且验证不可关闭。

```typescript TypeScript
import { LibertyNet } from "libertynet-sdk";

const ln = new LibertyNet();                 // 验证默认开启，且关不掉
const nodes = await ln.discovery.callable(); // 已验证、新鲜、可达、带签名

console.log(nodes.map((n) => `${n.did} ${n.endpoint} ${n.capabilities.join(",")}`));
```

```python
from libertynet import LibertyNet

ln = LibertyNet()                  # 验证默认开启，且无法关闭
nodes = ln.discovery.online()      # 只返回验证通过且新鲜的节点

for n in nodes:
    print(n.did, n.region, ",".join(n.capabilities))
```

`online()` 会拒绝任何身份验证不通过的记录，并丢弃太久没有心跳的节点。**没有开关可以关掉
这个检查**——一个会信任未经验证身份的客户端，我们不愿意发布。
